import pandas as pd

# Encoding-safe read for messy SMB vendor feeds
try:
    df = pd.read_csv(path, encoding="utf-8", dtype={"sku": str})
except UnicodeDecodeError:
    df = pd.read_csv(path, encoding="latin1", dtype={"sku": str})

df.columns = [c.strip().lower().replace(" ", "_") for c in df.columns]
df["sku"] = df["sku"].str.strip().str.upper()
